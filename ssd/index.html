async function submitTrial() { var email = document.getElementById('trial-email').value.trim(); var state = document.getElementById('trial-state').value; var status = document.getElementById('trial-status'); var btn = document.getElementById('trial-btn'); if (!email || !email.includes('@')) { status.textContent = 'Please enter a valid email'; return; } if (!state) { status.textContent = 'Please select a state'; return; } btn.textContent = 'Creating...'; btn.disabled = true; try { var res = await fetch('https://api.hazeydata.ai/ssd/create-trial', { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({email: email, state: state}) }); var data = await res.json(); if (data.api_key) { var htmlContent = '
' + '
Your API Key
' + '
' + data.api_key + '
' + '
Click to copy · Copied! Save this key — it will not be shown again.
Trial: ' + state + ' districts, 90 days, 10K calls/mo. API docs →
' + '
'; document.getElementById('trial-cta').innerHTML = htmlContent; } else { status.textContent = data.error || 'Something went wrong'; btn.textContent = 'Start Free Trial'; btn.disabled = false; } } catch(e) { status.textContent = 'Network error — please try again'; btn.textContent = 'Start Free Trial'; btn.disabled = false; } }
